cmake_minimum_required(VERSION 3.16)
project(BreadcasterAndCatcher LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "These input examples require Windows.")
endif()
add_library(input_protocol command.cpp)
target_include_directories(input_protocol PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(input_protocol PUBLIC cxx_std_17)
target_compile_definitions(input_protocol PUBLIC _WIN32_WINNT=0x0A00)
add_library(udp_sender "Breadcaster src/Broadcast_UDP.cpp")
target_include_directories(udp_sender PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Breadcaster src")
target_link_libraries(udp_sender PUBLIC input_protocol ws2_32)
add_library(udp_receiver "Catcher src/UDP_Receiver.cpp")
target_include_directories(udp_receiver PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Catcher src")
target_link_libraries(udp_receiver PUBLIC input_protocol ws2_32)
add_executable(breadcaster "Breadcaster src/main.cpp" "Breadcaster src/Input_Surveillance.cpp"
                           "Breadcaster src/Cursor_Surveillance.cpp")
target_link_libraries(breadcaster PRIVATE udp_sender user32)
add_executable(catcher "Catcher src/main.cpp" "Catcher src/Message_Interpreter.cpp")
target_link_libraries(catcher PRIVATE udp_receiver user32)
include(CTest)
if(BUILD_TESTING)
    add_executable(protocol-test tests/protocol_test.cpp)
    target_link_libraries(protocol-test PRIVATE udp_sender udp_receiver)
    add_test(NAME binary-protocol-loopback COMMAND protocol-test)
endif()
foreach(target input_protocol udp_sender udp_receiver breadcaster catcher protocol-test)
    if(TARGET ${target})
        set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /permissive-)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endforeach()
